theorem phasor_real_iff
    (t1 t2 : ℤ) (s : ℚ) (n3 : ℕ) (hn3 : 0 < n3) (r : ℝ) (hr : r ≠ 0) :
    (phasorValue t1 t2 s n3 r).im = 0 ↔ IsIntegral (phaseAddress t1 t2 s n3)

theorem integer_spin_phase_iff
    (t1 t2 s : ℤ) (n3 : ℕ) (hn3 : 0 < n3) :
    IsIntegral (phaseAddress t1 t2 (s : ℚ) n3) ↔
      IsIntegral (fordAddress t2 (s : ℚ) n3)

theorem integer_spin_parity
    (t1 t2 s : ℤ) (n3 : ℕ) (hn3 : 0 < n3) :
    Complex.exp (((Real.pi * (phaseAddress t1 t2 (s : ℚ) n3 : ℝ)) : ℂ) * Complex.I) =
      (-1 : ℂ) ^ (t1 * s) *
        Complex.exp (-(((Real.pi * (fordAddress t2 (s : ℚ) n3 : ℝ)) : ℂ) * Complex.I))

    phasorValue 1 1 0 3 r = phasorValue 1 1 3 3 r ∧

theorem reduced_det_mul
    (a c : ℤ) (q r : ℕ) (hq : 0 < q) (hr : 0 < r) :
    |det (rawValue a q) (rawValue c r)| * (gcdFactor a q : ℤ) * (gcdFactor c r : ℤ) =
      |a * (r : ℤ) - c * (q : ℤ)|

theorem reduced_det_quotient
    (a c : ℤ) (q r : ℕ) (hq : 0 < q) (hr : 0 < r) :
    |det (rawValue a q) (rawValue c r)| =
      |a * (r : ℤ) - c * (q : ℤ)| / ((gcdFactor a q : ℤ) * (gcdFactor c r : ℤ))

theorem row_law (N s : ℕ) (hN : 0 < N) (hs : s < N) :
    |det (rawValue (s : ℤ) N) (rawValue ((s + 1 : ℕ) : ℤ) N)| *
      (gcdFactor (s : ℤ) N : ℤ) * (gcdFactor ((s + 1 : ℕ) : ℤ) N : ℤ) = (N : ℤ)

theorem row_tangent_iff (N s : ℕ) (hN : 0 < N) (hs : s < N) :
    fareyAdj (rawValue (s : ℤ) N) (rawValue ((s + 1 : ℕ) : ℤ) N) ↔
      N = gcdFactor (s : ℤ) N * gcdFactor ((s + 1 : ℕ) : ℤ) N

theorem signed_column_law (u : ℤ) (n : ℕ) (hn : 1 ≤ n) :
    |det (rawValue u n) (rawValue u (n + 1))| *
      (gcdFactor u n : ℤ) * (gcdFactor u (n + 1) : ℤ) = |u|

theorem signed_column_tangent_iff (u : ℤ) (n : ℕ) (hn : 1 ≤ n) :
    fareyAdj (rawValue u n) (rawValue u (n + 1)) ↔
      |u| = (gcdFactor u n : ℤ) * (gcdFactor u (n + 1) : ℤ)

theorem zero_column_stall (n : ℕ) (hn : 1 ≤ n) :
    rawValue 0 n = rawValue 0 (n + 1) ∧
      det (rawValue 0 n) (rawValue 0 (n + 1)) = 0 ∧
      ¬fareyAdj (rawValue 0 n) (rawValue 0 (n + 1))

theorem reference_column_tangent_iff (s n : ℕ) (hs : 0 < s) (hn : 1 ≤ n) :
    fareyAdj (rawValue (s : ℤ) n) (rawValue (s : ℤ) (n + 1)) ↔ s ∣ n * (n + 1)

theorem unit_fraction_spine (n : ℕ) (hn : 1 ≤ n) :
    rawValue 1 n = unitFrac n ∧ fareyAdj (unitFrac n) (unitFrac (n + 1))

theorem shell_contact_parameterization
    (N : ℕ) (hN : 2 ≤ N) (s : ℚ) (hs0 : 0 ≤ s) (hsN : s ≤ (N : ℚ)) :
    IsIntegral (phaseAddress 1 1 s N) ↔
      ∃ m : ℕ, m ≤ N - 1 ∧ s = (m : ℚ) * (N : ℚ) / ((N : ℚ) - 1)

    (r : ℝ) (hr : r ≠ 0) :
    (phasorValue 1 1 s N r).im = 0 ↔
      ∃ m : ℕ, m ≤ N - 1 ∧ s = (m : ℚ) * (N : ℚ) / ((N : ℚ) - 1)

theorem shell_contact_address (N m : ℕ) (hN : 2 ≤ N) (hm : m ≤ N - 1) :
    fordAddress 1 ((m : ℚ) * (N : ℚ) / ((N : ℚ) - 1)) N =
      (m : ℚ) / ((N : ℚ) - 1)

theorem path_edge_trichotomy (m : ℕ) (x : Fin (m + 1) → ℚ) (k : Fin m) :
    let u := x k.castSucc
    let v := x k.succ
    (fareyAdj u v ∨ u = v ∨ IsBreak u v) ∧
      ¬(fareyAdj u v ∧ u = v) ∧
      ¬(fareyAdj u v ∧ IsBreak u v) ∧ ¬(u = v ∧ IsBreak u v)

    pathEdgeKind m x k = .tangent ↔ fareyAdj (x k.castSucc) (x k.succ)

    (m : ℕ) (x : Fin (m + 1) → ℚ) (k : Fin m)

      boundaryCheckQ (x k.succ) (kissQ (x k.castSucc) (x k.succ)) = true
